{{- /* Images. Page-bundle and asset images are resized to fit 1400×1400 as WebP and get width/height so the page doesn't jump while loading. An image on its own line becomes a numbered plate with its title as the caption: ![Alt text](kells.jpg "Caption, *italics* ok") */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $w := 0 -}}{{- $h := 0 -}} {{- if not $u.IsAbs -}} {{- with or (.PageInner.Resources.Get $u.Path) (resources.Get $u.Path) -}} {{- $img := . -}} {{- if and (eq .MediaType.MainType "image") (ne .MediaType.SubType "svg") (ne .MediaType.SubType "gif") -}} {{- if or (gt .Width 1400) (gt .Height 1400) -}} {{- $img = .Fit "1400x1400 webp q82" -}} {{- else -}} {{- $img = .Process "webp q82" -}} {{- end -}} {{- $w = $img.Width -}}{{- $h = $img.Height -}} {{- end -}} {{- $src = $img.RelPermalink -}} {{- end -}} {{- end -}} {{- if not .PlainText }}{{ warnf "Image without alt text in %s: %s" .Page.String .Destination }}{{ end -}} {{- $tag := printf `" $tag -}} {{- if .IsBlock -}} {{- else -}} {{ $tag | safeHTML }} {{- end -}}
