{{- /* A post's body as plain text for the search index, without verse numbers, footnote markers, footnotes' back-links or heading § anchors. */ -}} {{- $html := string .Content -}} {{- $html = replaceRE `[^<]*` "" $html -}} {{- $html = replaceRE `]*>[^<]*` "" $html -}} {{- $html = replaceRE `]*class="(anchor|footnote-backref)"[^>]*>[^<]*` "" $html -}} {{- $html = replaceRE `` "$0 " $html -}} {{- return ($html | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace) -}}
